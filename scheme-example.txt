Project student_application_system {
  database_type: "PostgreSQL"
}

//// ================= ENUMS =================

Enum user_role {
  admin
  agent
  student
  university
}

Enum doc_status {
  pending
  action_required
  verified
}

Enum app_status {
  draft
  submitted
  offered
  rejected
}

Enum kyc_status {
  pending
  verified
  rejected
}

Enum campus_type {
  main
  sub
  international
}

//// ================= USERS =================

Table users {
  id int [pk, increment]
  email varchar [unique, not null]
  password varchar
  role user_role
  is_deleted boolean [default: false]
  created_at timestamp
  updated_at timestamp
}

//// ================= UNIVERSITIES =================

Table universities {
  id int [pk, increment]
  user_id int [ref: > users.id]
  name varchar
  is_deleted boolean [default: false]
  created_at timestamp
}

//// ================= CAMPUSES =================

Table campuses {
  id int [pk, increment]
  university_id int [ref: > universities.id]
  name varchar
  location varchar
  address text
  campus_type campus_type
  established_year int
  total_area varchar
  description text
  cover_image varchar
  is_deleted boolean [default: false]
}

Table campus_departments {
  id int [pk, increment]
  campus_id int [ref: > campuses.id]
  name varchar
}

Table campus_faculties {
  id int [pk, increment]
  campus_id int [ref: > campuses.id]
  name varchar
}

//// ================= PROGRAMS =================

Table programs {
  id int [pk, increment]
  university_id int [ref: > universities.id]
  name varchar
  category varchar
  tuition_fees decimal
  agent_commission decimal
  program_length varchar
  study_type varchar
  intake_date date
  description text
  admission_requirements text
  prospects text
  competency_model text
  professional_skills text
  management_skills text
  is_deleted boolean [default: false]
}

//// 🔥 PROGRAM ↔ CAMPUS (M:N)

Table program_campuses {
  id int [pk, increment]
  program_id int [ref: > programs.id]
  campus_id int [ref: > campuses.id]
}

//// ================= STUDENTS =================

Table students {
  id int [pk, increment]
  user_id int [ref: > users.id]
  full_name varchar
  phone varchar
  dob date
  gender varchar
  country varchar
  nationality varchar
  parent_email varchar
  parent_phone varchar
  profile_image varchar
  passport_url varchar
  added_by int [ref: > users.id, null]
  is_deleted boolean [default: false]
  created_at timestamp
}

Table student_qualifications {
  id int [pk, increment]
  student_id int [ref: > students.id]
  qualification varchar
  institution_name varchar
  gpa decimal
  is_deleted boolean [default: false]
}

Table student_experiences {
  id int [pk, increment]
  student_id int [ref: > students.id]
  company_name varchar
  role varchar
  years int
  is_deleted boolean [default: false]
}

//// ================= AGENTS =================

Table agents {
  id int [pk, increment]
  user_id int [ref: > users.id]
  agency_name varchar
  contact_person_name varchar
  gender varchar
  country varchar
  profession varchar
  website varchar
  nationality varchar
  address text
  profile_image varchar
  passport_url varchar
  id_card_front varchar
  id_card_back varchar
  experience varchar
  kyc_status kyc_status
  is_deleted boolean [default: false]
}

//// ================= DOCUMENTS =================

Table documents {
  id int [pk, increment]
  student_id int [ref: > students.id]
  type varchar
  url varchar
  status doc_status
  added_by int [ref: > users.id]
  is_deleted boolean [default: false]
  created_at timestamp
}

//// ================= APPLICATIONS =================

Table applications {
  id int [pk, increment]
  student_id int [ref: > students.id]
  agent_id int [ref: > agents.id, null]
  campus_id int [ref: > campuses.id]
  status app_status
  academic_session varchar
  is_deleted boolean [default: false]
  created_at timestamp
}

//// 🔥 STRICT PROGRAM + CAMPUS LINK

Table application_programs {
  id int [pk, increment]
  application_id int [ref: > applications.id]
  program_campus_id int [ref: > program_campuses.id]
  status varchar
}

//// APPLICATION DOCUMENTS

Table application_documents {
  id int [pk, increment]
  application_id int [ref: > applications.id]
  document_id int [ref: > documents.id]
}

//// ================= PAYMENTS =================

Table payments {
  id int [pk, increment]
  application_id int [ref: > applications.id]
  student_id int [ref: > students.id]
  amount decimal
  status varchar
  payment_method varchar
  transaction_id varchar
  created_at timestamp
}
